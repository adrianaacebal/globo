// Builds the Express app. Does not call listen — that lives in server.js.
const express = require("express");
const path = require("path");
const logger = require("./middleware/logger");
const { notFound, errorHandler } = require("./middleware/errors");
const restaurantsRouter = require("./routes/restaurants");
const dishesRouter = require("./routes/dishes");
const ordersRouter = require("./routes/orders");
const tasksRouter = require("./routes/tasks");

const app = express();

// Middleware runs before the routes
app.use(express.json());
app.use((req, res, next) => {
  // Express 5 deja req.body en undefined si no llega body: asi los controladores siempre tienen un objeto
  if (req.body === undefined) req.body = {};
  next();
});
app.use(logger);

// API
app.get("/health", (req, res) => res.json({ status: "ok" }));

// Session 13 lab: reflects the URL segment -> /echo/hi gives { "echo": "hi" }
app.get("/echo/:msg", (req, res) => res.json({ echo: req.params.msg }));

// Session 13 lab: Tasks API (thin router -> tasksController)
app.use("/tasks", tasksRouter);

// Session 13 lab: forces an error to prove the error handler (registered last) catches it
app.get("/boom", (req, res, next) => {
  next(new Error("Boom! Forced failure"));
});

app.use("/api/restaurants", restaurantsRouter);
app.use("/api/dishes", dishesRouter);
app.use("/", ordersRouter); // /login, /menu, /order, /order/total, /orders

// El codigo del servidor y los datos (usuarios y contrasenas) no se sirven al navegador
const PRIVATE = /^\/(src|data|node_modules|postman|package(-lock)?\.json)(\/|$)/i;
app.use((req, res, next) => (PRIVATE.test(req.path) ? notFound(req, res) : next()));

// The pages now live in html/, so the home URL sends you to the homepage
app.get("/", (req, res) => res.redirect("/html/index.html"));

// Serves the HTML pages (html/login.html, html/index.html...) from the project folder
app.use(express.static(path.join(__dirname, "..")));

// Anything else -> 404 JSON, then the centralised error handler (registered last)
app.use(notFound);
app.use(errorHandler);

module.exports = app;

// Logica de /login
const store = require("../store");
const { newSession } = require("../sessions");
const { DISCOUNT_PERCENT, newDiscountCode } = require("../pricing");

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

// Errores del registro, todos a la vez
function registrationErrors({ name, email, password }) {
  const errors = {};
  if (!name) errors.name = "name is required";
  else if (name.length > 60) errors.name = "name must be at most 60 characters";
  if (!email) errors.email = "email is required";
  else if (!EMAIL_RE.test(email)) errors.email = "email is not valid";
  if (!password) errors.password = "password is required";
  else if (password.length < 6) errors.password = "password must be at least 6 characters";
  return Object.keys(errors).length ? errors : null;
}

// POST /login — si el email existe comprueba nombre y contrasena; si no, registra al usuario nuevo
function login(req, res) {
  const name = String(req.body.name ?? "").trim();
  const email = String(req.body.email ?? "").trim().toLowerCase();
  const password = String(req.body.password ?? "");

  // El email no distingue mayusculas (Ana@Mail.com es el mismo que ana@mail.com)
  const user = email ? store.users.find((u) => u.email.toLowerCase() === email) : null;

  if (user) {
    const sameName = user.name.trim().toLowerCase() === name.toLowerCase();
    if (!sameName || user.password !== password) {
      return res.status(401).json({ error: "This email is already registered. Incorrect name or password" });
    }
    // Si todavia no ha usado su codigo de bienvenida, se le vuelve a dar
    const pending = user.discountCode && !user.discountUsed;
    return res.json({
      name: user.name,
      email: user.email,
      token: newSession(user),
      discountCode: pending ? user.discountCode : undefined,
      discountPercent: pending ? DISCOUNT_PERCENT : undefined,
    });
  }

  const errors = registrationErrors({ name, email, password });
  if (errors) {
    return res.status(400).json({ error: Object.values(errors).join(". "), errors });
  }

  // Primera vez: se crea la cuenta y se le regala un codigo de descuento
  const newUser = store.users.create({
    name,
    email,
    password,
    discountCode: newDiscountCode(),
    discountUsed: false,
    orders: [],
  });
  res.status(201).json({
    name: newUser.name,
    email: newUser.email,
    token: newSession(newUser),
    firstTime: true,
    discountCode: newUser.discountCode,
    discountPercent: DISCOUNT_PERCENT,
  });
}

module.exports = { login };

// Logica de /api/dishes (platos de la carta de cada marca) y de /menu
const store = require("../store");
const { validateDish, validateDishQuery } = require("../validators/dishes");
const { normalize, brandKeyFromName, round2 } = require("../utils");

// Aplica los filtros de la query: ?q=&brand=&category=&minPrice=&maxPrice=
function filterDishes(list, f) {
  return list.filter(
    (d) =>
      (!f.brand || d.brand === f.brand) &&
      (!f.category || d.category === f.category) &&
      (f.minPrice === undefined || d.price >= f.minPrice) &&
      (f.maxPrice === undefined || d.price <= f.maxPrice) &&
      (!f.q || normalize(d.name).includes(normalize(f.q)))
  );
}

// Una marca no puede tener dos platos con el mismo nombre (exceptId: el plato que se edita)
function isDuplicate({ brand, name }, exceptId) {
  return Boolean(
    store.dishes.find((d) => d.id !== exceptId && d.brand === brand && normalize(d.name) === normalize(name))
  );
}

const notFound = (res) => res.status(404).json({ error: "Dish not found" });
const duplicate = (res) =>
  res.status(409).json({ errors: { name: "This brand already has a dish with that name" } });

// { count, averagePrice } de una lista de platos
function priceSummary(list) {
  const sum = list.reduce((acc, d) => acc + d.price, 0);
  return { count: list.length, averagePrice: list.length ? round2(sum / list.length) : 0 };
}

function groupSummary(list, field) {
  const groups = {};
  for (const dish of list) (groups[dish[field]] ||= []).push(dish);
  return Object.fromEntries(Object.entries(groups).map(([key, items]) => [key, priceSummary(items)]));
}

// GET /api/dishes -> 200 [ ... ]
function listDishes(req, res) {
  const { value: filters, errors } = validateDishQuery(req.query);
  if (errors) return res.status(400).json({ errors });
  res.json(filterDishes(store.dishes.all(), filters));
}

// GET /api/dishes/stats (acepta los mismos filtros)
function dishStats(req, res) {
  const { value: filters, errors } = validateDishQuery(req.query);
  if (errors) return res.status(400).json({ errors });

  const list = filterDishes(store.dishes.all(), filters);
  const prices = list.map((d) => d.price);
  res.json({
    total: list.length,
    averagePrice: priceSummary(list).averagePrice,
    minPrice: list.length ? Math.min(...prices) : null,
    maxPrice: list.length ? Math.max(...prices) : null,
    byBrand: groupSummary(list, "brand"),
    byCategory: groupSummary(list, "category"),
  });
}

// GET /api/dishes/:id
function getDish(req, res) {
  const dish = store.dishes.findById(req.id);
  if (!dish) return notFound(res);
  res.json(dish);
}

// POST /api/dishes -> 201 (409 si la marca ya tiene un plato con ese nombre)
function createDish(req, res) {
  const { value, errors } = validateDish(req.body);
  if (errors) return res.status(400).json({ errors });
  if (isDuplicate(value)) return duplicate(res);
  res.status(201).json(store.dishes.create(value));
}

// PUT /api/dishes/:id
function replaceDish(req, res) {
  if (!store.dishes.findById(req.id)) return notFound(res);
  const { value, errors } = validateDish(req.body);
  if (errors) return res.status(400).json({ errors });
  if (isDuplicate(value, req.id)) return duplicate(res);
  res.json(store.dishes.replace(req.id, value));
}

// PATCH /api/dishes/:id
function updateDish(req, res) {
  const current = store.dishes.findById(req.id);
  if (!current) return notFound(res);
  const { value, errors } = validateDish(req.body, { partial: true });
  if (errors) return res.status(400).json({ errors });
  if (isDuplicate({ ...current, ...value }, req.id)) return duplicate(res);
  res.json(store.dishes.update(req.id, value));
}

// DELETE /api/dishes/:id -> 204
function deleteDish(req, res) {
  if (!store.dishes.remove(req.id)) return notFound(res);
  res.status(204).end();
}

// GET /menu?restaurant=McDonald's -> carta completa de la marca (la usa order.html)
function getMenu(req, res) {
  const { restaurant } = req.query;
  if (typeof restaurant !== "string" || !restaurant.trim()) {
    return res.status(400).json({ errors: { restaurant: "restaurant is required" } });
  }
  const brand = brandKeyFromName(restaurant);
  if (!brand) return res.status(404).json({ error: "Restaurant not found" });
  res.json(store.dishes.all().filter((d) => d.brand === brand));
}

module.exports = { listDishes, dishStats, getDish, createDish, replaceDish, updateDish, deleteDish, getMenu };

// Logica de /order, /order/total y /orders (pedidos del usuario conectado)
const store = require("../store");
const { userFromRequest } = require("../sessions");
const { ORDER_TYPES } = require("../constants");
const { normalize, pickOption, brandKeyFromName, round2 } = require("../utils");
const { DISCOUNT_PERCENT, deliveryFeePercent, isValidCode } = require("../pricing");

// Respuesta 400 con todos los errores: "error" (texto para el cliente actual) + "errors" por campo
const badRequest = (res, errors) => res.status(400).json({ error: Object.values(errors).join(". "), errors });

// POST /order — comprueba que la marca tiene local en esa zona (requireUser va antes)
function createOrder(req, res) {
  const { name, email } = req.user;
  const { restaurant, location } = req.body;
  const type = pickOption(ORDER_TYPES, req.body.type);

  const errors = {};
  if (!restaurant) errors.restaurant = "Restaurant is required";
  if (!location) errors.location = "Location is required";
  if (!type) errors.type = `type must be one of: ${ORDER_TYPES.join(", ")}`;
  if (Object.keys(errors).length) return badRequest(res, errors);

  const brand = brandKeyFromName(restaurant);
  const brandLocals = brand ? store.restaurants.all().filter((r) => r.brand === brand) : [];
  if (brandLocals.length === 0) {
    return res.status(404).json({ error: "Restaurant not found" });
  }

  const local = brandLocals.find((r) => normalize(r.area) === normalize(location));
  if (!local) {
    const allowed = [...new Set(brandLocals.map((r) => r.area))];
    return badRequest(res, {
      location: `The location ${location} does not exist for ${restaurant}. Available locations: ${allowed.join(", ")}`,
    });
  }

  res.status(201).json({ message: "Order created", order: { name, email, type, restaurant: local.name, area: local.area } });
}

// Calcula el precio de un pedido. Lo usan /order/total y /orders.
// Devuelve { status, error } si algo esta mal, o el desglose del precio.
function computePrice({ restaurant, items, type, travelMinutes, discountCode }, user) {
  const brand = brandKeyFromName(restaurant);
  if (!brand) return { status: 404, error: "Restaurant not found" };

  if (!Array.isArray(items) || items.length === 0) {
    return { status: 400, error: "Choose at least one dish" };
  }

  const menu = store.dishes.all().filter((d) => d.brand === brand);
  let subtotal = 0;
  for (const item of items) {
    const menuItem = menu.find((d) => d.name === item?.dish);
    if (!menuItem) return { status: 404, error: `Dish not found: ${item?.dish}` };

    const qty = Number(item.quantity);
    if (!Number.isInteger(qty) || qty < 1) {
      return { status: 400, error: "Quantity must be 1 or more" };
    }
    subtotal += menuItem.price * qty;
  }

  // 1) Envio a domicilio: % sobre el precio de la comida
  const feePercent = deliveryFeePercent(type, travelMinutes);
  const deliveryFee = (subtotal * feePercent) / 100;

  // 2) Despues, el codigo de bienvenida: 10% menos sobre comida + envio (solo el del propio usuario)
  let discount = 0;
  let discountError;
  if (discountCode) {
    if (isValidCode(discountCode, user)) {
      discount = ((subtotal + deliveryFee) * DISCOUNT_PERCENT) / 100;
    } else {
      discountError = "This discount code is not valid or has already been used";
    }
  }

  return {
    // Importes redondeados a centimos
    subtotal: round2(subtotal),
    discount: round2(discount),
    deliveryFee: round2(deliveryFee),
    deliveryFeePercent: feePercent,
    total: round2(subtotal + deliveryFee - discount),
    discountError,
  };
}

// POST /order/total — solo calcula, no guarda nada (el login es opcional: hace falta para el codigo)
function calculateTotal(req, res) {
  const price = computePrice(req.body, userFromRequest(req));
  if (price.error) return res.status(price.status).json({ error: price.error });
  res.json(price);
}

// POST /orders — confirma el pedido y lo guarda en el historial (requireUser va antes).
// Hay que volver a escribir la contrasena.
function saveOrder(req, res) {
  const user = req.user;
  const { password, restaurant, location, type, items, payment } = req.body;
  if (password !== user.password) {
    return res.status(401).json({ error: "Incorrect password" });
  }

  const price = computePrice(req.body, user);
  if (price.error) return res.status(price.status).json({ error: price.error });

  const order = {
    id: Date.now(),
    date: new Date().toISOString(),
    restaurant,
    location,
    type,
    payment,
    items: items.map(({ dish, quantity }) => ({ dish, quantity: Number(quantity) })),
    subtotal: price.subtotal,
    discount: price.discount,
    deliveryFee: price.deliveryFee,
    deliveryFeePercent: price.deliveryFeePercent,
    total: price.total,
  };

  store.users.update(user.id, {
    orders: [...(user.orders || []), order],
    // El codigo de descuento solo vale una vez
    discountUsed: Boolean(user.discountUsed || price.discount > 0),
  });

  res.status(201).json({ message: "Order placed", order });
}

// GET /orders — historial del usuario conectado (el mas nuevo primero)
function listOrders(req, res) {
  res.json((req.user.orders || []).slice().reverse());
}

module.exports = { createOrder, calculateTotal, saveOrder, listOrders };

// Logica de /api/restaurants (locales de cada marca)
const store = require("../store");
const { validateRestaurant, validateRestaurantQuery } = require("../validators/restaurants");
const { normalize } = require("../utils");

// Aplica los filtros de la query: ?q=&brand=&area=&type=
// q busca en el nombre y en la zona ("retiro" encuentra "Telepizza Atocha", que esta en Retiro)
function filterRestaurants(list, filters) {
  const q = normalize(filters.q);
  return list.filter(
    (r) =>
      (!filters.brand || r.brand === filters.brand) &&
      (!filters.area || r.area === filters.area) &&
      (!filters.type || r.type === filters.type) &&
      (!q || normalize(r.name + r.area).includes(q))
  );
}

// { Centro: 4, Retiro: 2, ... }
function countBy(list, field) {
  return list.reduce((acc, item) => {
    acc[item[field]] = (acc[item[field]] || 0) + 1;
    return acc;
  }, {});
}

const notFound = (res) => res.status(404).json({ error: "Restaurant not found" });

// GET /api/restaurants  (?q=&brand=&area=&type=) -> 200 [ ... ] (lista vacia si no hay coincidencias)
function listRestaurants(req, res) {
  const { value: filters, errors } = validateRestaurantQuery(req.query);
  if (errors) return res.status(400).json({ errors });
  res.json(filterRestaurants(store.restaurants.all(), filters));
}

// GET /api/restaurants/stats  (acepta los mismos filtros)
function restaurantStats(req, res) {
  const { value: filters, errors } = validateRestaurantQuery(req.query);
  if (errors) return res.status(400).json({ errors });

  const list = filterRestaurants(store.restaurants.all(), filters);
  res.json({
    total: list.length,
    byBrand: countBy(list, "brand"),
    byArea: countBy(list, "area"),
    byType: countBy(list, "type"),
  });
}

// Rutas que no existen -> 404 en JSON (se registra despues de todas las rutas)
function notFound(req, res) {
  res.status(404).json({ error: `Route ${req.method} ${req.originalUrl} not found` });
}

// Manejador de errores centralizado (se registra el ultimo)
function errorHandler(err, req, res, next) {
  // JSON mal escrito en el body -> es culpa del cliente
  if (err.type === "entity.parse.failed") {
    return res.status(400).json({ error: "Invalid JSON in request body" });
  }
  // Otros errores del cliente que lanza Express (p. ej. 413 body demasiado grande)
  if (err.status >= 400 && err.status < 500) {
    return res.status(err.status).json({ error: err.message });
  }
  console.error(err);
  res.status(500).json({ error: "Internal server error" });
}

module.exports = { notFound, errorHandler };

// Se usa con router.param("id", validateId): comprueba :id antes de llegar al controlador.

function validateId(req, res, next, value) {
  if (!/^[1-9]\d*$/.test(value)) {
    return res.status(400).json({ error: "id must be a positive integer" });
  }
  req.id = Number(value);
  next();
}

// Reglas de precio compartidas: codigo de bienvenida y gastos de envio
const crypto = require("crypto");
const store = require("./store");

const DISCOUNT_PERCENT = 10; // % de descuento del codigo de bienvenida

// Gastos de envio segun lo que tarda la moto (sin contar la cocina):
//   hasta 15 min -> 5% del precio de la comida · mas de 15 min -> 7%
const SHORT_TRIP_MAX_MINUTES = 15;
const SHORT_TRIP_FEE_PERCENT = 5;
const LONG_TRIP_FEE_PERCENT = 7;

// Porcentaje de envio para un pedido (0 si es para recoger)
function deliveryFeePercent(type, travelMinutes) {
  if (type !== "delivery") return 0;
  const mins = Number(travelMinutes);
  if (Number.isFinite(mins) && mins <= SHORT_TRIP_MAX_MINUTES) return SHORT_TRIP_FEE_PERCENT;
  return LONG_TRIP_FEE_PERCENT;
}

// "GLOBO-3F9A1C" (se repite si por casualidad ya existe)
function newDiscountCode() {
  let code;
  do {
    code = "GLOBO-" + crypto.randomBytes(3).toString("hex").toUpperCase();
  } while (store.users.find((u) => u.discountCode === code));
  return code;
}

===== src/sessions.js =====
// Sesiones: al hacer login el servidor da un "token" (texto aleatorio) que el navegador guarda
// y manda en la cabecera Authorization: Bearer <token>. Asi sabemos quien es.
// Se guardan en memoria: si se reinicia el servidor hay que volver a hacer login.
const crypto = require("crypto");
const store = require("./store");

const sessions = new Map(); // token -> email

function newSession(user) {
  const token = crypto.randomBytes(24).toString("hex");
  sessions.set(token, user.email);
  return token;
}

// Devuelve el usuario conectado, o null si no hay token valido
function userFromRequest(req) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : "";
  const email = sessions.get(token);
  return email ? store.users.find((u) => u.email === email) : null;
}

module.exports = { newSession, userFromRequest };

// Capa de datos. Los controladores SOLO hablan con estas funciones, nunca con los arrays.
// Para pasar a una base de datos real basta con reescribir este archivo con la misma interfaz:
//   all() · findById(id) · find(predicate) · create(data) · replace(id, data) · update(id, changes) · remove(id)
const fs = require("fs");
const path = require("path");
const seed = require("./datos");

const copy = (item) => structuredClone(item);

// Crea una coleccion en memoria. onChange(items) se llama tras cada escritura (para guardar en disco).
function createCollection(initial = [], onChange = () => {}) {
  let nextId = initial.reduce((max, item) => Math.max(max, item.id || 0), 0) + 1;
  const items = initial.map((item) => (item.id ? copy(item) : { id: nextId++, ...copy(item) }));

  const indexOf = (id) => items.findIndex((item) => item.id === id);

  return {
    all() {
      return items.map(copy);
    },

    findById(id) {
      const item = items.find((i) => i.id === id);
      return item ? copy(item) : null;
    },

    find(predicate) {
      const item = items.find(predicate);
      return item ? copy(item) : null;
    },

    // El id siempre lo pone el servidor, aunque venga uno en data
    create(data) {
      const { id: _ignored, ...fields } = data;
      const item = { id: nextId++, ...copy(fields) };
      items.push(item);
      onChange(items);
      return copy(item);
    },

    // PUT: sustituye todos los campos (menos el id)
    replace(id, data) {
      const index = indexOf(id);
      if (index === -1) return null;
      const { id: _ignored, ...fields } = data;
      items[index] = { id, ...copy(fields) };
      onChange(items);
      return copy(items[index]);
    },

    // PATCH: cambia solo los campos recibidos
    update(id, changes) {
      const index = indexOf(id);
      if (index === -1) return null;
      const { id: _ignored, ...fields } = changes;
      items[index] = { ...items[index], ...copy(fields) };
      onChange(items);
      return copy(items[index]);
    },

    remove(id) {
      const index = indexOf(id);
      if (index === -1) return false;
      items.splice(index, 1);
      onChange(items);
      return true;
    },
  };
}

const { BRAND_KEYS } = require("./constants");

// Normaliza texto para comparar: minusculas, sin tildes, sin espacios ni apostrofes
// ("McDonald's" -> "mcdonalds", "Chamberí" -> "chamberi")
function normalize(text) {
  return String(text ?? "")
    .toLowerCase()
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .replace(/[^a-z0-9]/g, "");
}

// Busca un valor en una lista fija sin importar mayusculas/tildes y devuelve el valor oficial
// pickOption(AREAS, "chamberi") -> "Chamberí"   ·   pickOption(AREAS, "Paris") -> undefined
function pickOption(options, value) {
  const wanted = normalize(value);
  return options.find((option) => normalize(option) === wanted);
}

// Marca a partir de un nombre de local: "Telepizza Gran Vía" -> "telepizza"
function brandKeyFromName(name) {
  const clean = normalize(name);
  return clean ? BRAND_KEYS.find((key) => clean.startsWith(key)) : undefined;
}

const round2 = (n) => Math.round(n * 100) / 100;

module.exports = { normalize, pickOption, brandKeyFromName, round2 };

===== src/validators/common.js =====
// Validacion comun de body y query (la usan los validators de cada recurso). Devuelve TODOS los errores a la vez: { value, errors }
// - value: solo los campos conocidos y ya limpios (los desconocidos y el id se ignoran)
// - errors: { campo: "mensaje" } o null si todo esta bien
const { pickOption, round2 } = require("../utils");

// ---------- Reglas por tipo de campo ----------
function checkField(field, rule, raw) {
  if (Array.isArray(raw)) return { error: `${field} must be a single value` };

  if (rule.type === "text") {
    if (typeof raw !== "string") return { error: `${field} must be text` };
    const text = raw.trim();
    if (text.length < rule.min || text.length > rule.max) {
      return { error: `${field} must be between ${rule.min} and ${rule.max} characters` };
    }
    return { value: text };
  }

  if (rule.type === "number") {
    // Se aceptan numeros y textos numericos ("7.5"), porque los formularios mandan texto
    const n = typeof raw === "number" ? raw : typeof raw === "string" && raw.trim() !== "" ? Number(raw) : NaN;
    if (!Number.isFinite(n)) return { error: `${field} must be a number` };
    if (n < rule.min || n > rule.max) return { error: `${field} must be between ${rule.min} and ${rule.max}` };
    return { value: rule.decimals !== undefined ? round2(n) : n };
  }

  if (rule.type === "option") {
    const option = typeof raw === "string" ? pickOption(rule.options, raw) : undefined;
    if (!option) return { error: `${field} must be one of: ${rule.options.join(", ")}` };
    return { value: option };
  }

  throw new Error(`Unknown rule type ${rule.type}`);
}

const isEmpty = (raw) => raw === null || (typeof raw === "string" && raw.trim() === "");
const isPlainObject = (x) => x !== null && typeof x === "object" && !Array.isArray(x);

// partial = true en PATCH: los campos que no vienen no son obligatorios
function validateBody(schema, body, { partial = false } = {}) {
  if (!isPlainObject(body)) return { value: {}, errors: { body: "Body must be a JSON object" } };

  const value = {};
  const errors = {};
  for (const [field, rule] of Object.entries(schema)) {
    const raw = body[field];
    if (raw === undefined && partial) continue;
    if (raw === undefined || isEmpty(raw)) {
      errors[field] = `${field} is required`;
      continue;
    }
    const result = checkField(field, rule, raw);
    if (result.error) errors[field] = result.error;
    else value[field] = result.value;
  }

  if (partial && Object.keys(value).length === 0 && Object.keys(errors).length === 0) {
    errors.body = `Send at least one of: ${Object.keys(schema).join(", ")}`;
  }
  return { value, errors: Object.keys(errors).length ? errors : null };
}

// En la query todo es opcional; los parametros desconocidos se ignoran
function validateQuery(schema, query) {
  const value = {};
  const errors = {};
  for (const [field, rule] of Object.entries(schema)) {
    const raw = query[field];
    if (raw === undefined || raw === "") continue;
    const result = checkField(field, rule, raw);
    if (result.error) errors[field] = result.error;
    else value[field] = result.value;
  }
  return { value, errors };
}

module.exports = { validateBody, validateQuery };
